\begin{enumerate}

\item Within Lecture~13, we derived the following equation
\begin{equation*}
    \rho\frac{\partial v_{i}}{\partial t}-\frac{\partial T_{ij}}{\partial x_{j}}=0,
\end{equation*}
for an elastic body. Here $\rho$ is the referential density, $v_{i} = \frac{\partial \varphi_{i}}{\partial t}$ the velocity with $\bphi$ the motion, and $\mathbf{T}$ the first Piola--Kirchhoff stress tensor.

\begin{enumerate}
     \item 
     For an arbitrary sub-body, $U$, of the reference body, $M$, show that the equality
    \begin{equation*}
        \frac{\ddns}{\ddns t}\int_{U}\rho v_{i}\dd^{3}\mathbf{x}=\int_{\partial U}T_{ij}\hat{n}_{j}\dd S,
    \end{equation*}
    holds, where $\partial U$ is the boundary of $U$. What is the physical significance of this result? In particular, what does the vector $T_{ij}\hat{n}_{j}$ represent?


    \item
    The angular momentum of the sub-body $U$ about the origin is given by
    \begin{equation*}
        \int_{U}\rho\epsilon_{ijk}\varphi_{j}v_{k}\dd^{3}\mathbf{x}.
    \end{equation*}
    Show that the angular momentum of every such sub-body changes only through the torque exerted by the tractions on its boundary if and only if $\mathbf{F}\mathbf{T}^{T}$ is a symmetric tensor. Defining the second Piola--Kirchhoff stress, $\mathbf{S}$, through $\mathbf{T}=\mathbf{F}\mathbf{S}$, conclude that $\mathbf{S}$ is symmetric.

   
    

\end{enumerate}

\item Continuing from the previous question, we define the energy density of the body by
\begin{equation*}
    E=\frac{1}{2}\rho v_{i}v_{i}+W(\mathbf{x},\mathbf{F}),
\end{equation*}
where $W$ is the strain energy function, so that $T_{ij}=\frac{\partial W}{\partial F_{ij}}$. Show that
\begin{equation*}
    \frac{\partial E}{\partial t}+\frac{\partial s_{i}}{\partial x_{i}}=0,
\end{equation*}
where $s_{i}$ is a vector field to be determined. By integrating this identity over an arbitrary sub-body, $U$, explain the physical meaning of $s_{i}$, and hence show that the total energy of the body is conserved, using the traction-free boundary condition $T_{ij}\hat{n}_{j}=0$ on $\partial M$.

\item The Lagrangian density for the linearised motion of an elastic body subject to a stress glut is given by
\begin{equation*}
    \mathcal{L}=\frac{1}{2}\rho\frac{\partial u_{i}}{\partial t}\frac{\partial u_{i}}{\partial t}-\frac{1}{2}A_{ijkl}\frac{\partial u_{i}}{\partial x_{j}}\frac{\partial u_{k}}{\partial x_{l}}+\overline{S}_{ij}\frac{\partial u_{i}}{\partial x_{j}},
\end{equation*}
where $\mathbf{u}$ is the displacement vector, $A_{ijkl}$ the elastic tensor, and $\overline{\mathbf{S}}$ the stress glut. Using this expression, write down the equations of motion and natural boundary conditions; here you may assume the standard form of the Euler--Lagrange equations.

\item The principle of material frame indifference shows that the strain energy function, $W$, of an elastic body depends on the deformation gradient, $\mathbf{F}$, only through the right Cauchy--Green deformation tensor, $\mathbf{C}=\mathbf{F}^{T}\mathbf{F}$, and hence we can write $W(\mathbf{x},\mathbf{F})=U(\mathbf{x},\mathbf{C})$, for some auxiliary function, $U$. The elastic tensor, $A_{ijkl}$, is defined by
\begin{equation*}
    A_{ijkl}=\frac{\partial^{2}W}{\partial F_{ij}\partial F_{kl}},
\end{equation*}
where the partial derivative on the right-hand side is evaluated at the equilibrium value $\mathbf{F}=\mathbf{I}$. Assuming that the equilibrium is stress-free, show that the elastic tensor possesses the symmetries $A_{ijkl}=A_{jikl}=A_{ijlk}=A_{klij}$.
How many independent components does such an elastic tensor have? Do these symmetries remain if there is a non-zero equilibrium stress?

\item In lectures we found the stress glut for an idealised fault surface $\Sigma$ to be $\overline{S}_{ij}=A_{ijkl}w_{k}\hat{n}_{l}\delta_{\Sigma}$, where $\mathbf{w}$ is the displacement discontinuity vector, which is tangential to $\Sigma$, $\hat{\mathbf{n}}$ is the unit normal to $\Sigma$, and $\delta_{\Sigma}$ is the Dirac surface distribution for $\Sigma$.
\begin{enumerate}
    \item Consider a planar fault with unit normal $\hat{\mathbf{n}}$ within an isotropic body. Show that
    \begin{equation*}
        \overline{S}_{ij}=\mu(w_{i}\hat{n}_{j}+w_{j}\hat{n}_{i})\delta_{\Sigma},
    \end{equation*}
    and comment on the absence of $\lambda$. Verify that this stress glut is symmetric and has zero trace.
    \hardp Consider the linearised equations of motion in a body $M$ subject to a body force $f_{i}$ and an applied traction $t_{i}$ on $\partial M$,
    \begin{equation*}
        \rho\frac{\partial^{2}u_{i}}{\partial t^{2}}-\frac{\partial}{\partial x_{j}}\left(A_{ijkl}\frac{\partial u_{k}}{\partial x_{l}}\right)=f_{i}\quad\text{in }M,\qquad A_{ijkl}\frac{\partial u_{k}}{\partial x_{l}}\hat{n}_{j}=t_{i}\quad\text{on }\partial M,
    \end{equation*}
    with the body at rest prior to $t=0$. The solution of this problem can be expressed in terms of a Green function, $G_{ij}(\mathbf{x},t;\mathbf{x}',t')$, in the form
    \begin{multline*}
        u_{i}(\mathbf{x},t)=\int_{0}^{t}\int_{M}G_{ij}(\mathbf{x},t;\mathbf{x}',t')f_{j}(\mathbf{x}',t')\dd^{3}\mathbf{x}'\dd t' \\
        +\int_{0}^{t}\int_{\partial M}G_{ij}(\mathbf{x},t;\mathbf{x}',t')t_{j}(\mathbf{x}',t')\dd S'\dd t',
    \end{multline*}
    where the detailed form of $G_{ij}$ will not be needed. Recall that a stress glut, $\overline{S}_{ij}$, enters the linearised equations of motion as an effective body force $f_{i}=-\partial\overline{S}_{ij}/\partial x_{j}$ together with an effective traction $t_{i}=\overline{S}_{ij}\hat{n}_{j}$, where $\hat{\mathbf{n}}$ is the outward unit normal to $\partial M$. Note that within the integrals above $f_{j}$, $t_{j}$ and $\hat{\mathbf{n}}$ are all evaluated at the integration point $\mathbf{x}'$. Show that the resulting displacement can be written as a single integral over the stress glut,
    \begin{equation*}
        u_{i}(\mathbf{x},t)=\int_{0}^{t}\int_{M}\frac{\partial G_{ij}}{\partial x_{k}'}(\mathbf{x},t;\mathbf{x}',t')\overline{S}_{jk}(\mathbf{x}',t')\dd^{3}\mathbf{x}'\dd t'.
    \end{equation*}
    \hardp Using the result of part (a), show that the displacement generated by slip on a planar fault within an isotropic body is
    \begin{equation*}
        u_{i}(\mathbf{x},t)=\int_{0}^{t}\int_{\Sigma}\mu\left[\frac{\partial G_{ij}}{\partial x_{k}'}+\frac{\partial G_{ik}}{\partial x_{j}'}\right](\mathbf{x},t;\mathbf{x}',t')\,w_{j}(\mathbf{x}',t')\hat{n}_{k}(\mathbf{x}')\dd S'\dd t',
    \end{equation*}
    and comment on the physical content of this expression. (The normal $\hat{\mathbf{n}}$ is constant for a planar fault, but it has been written as a function of $\mathbf{x}'$ because the result holds unchanged for a general fault surface.)
\end{enumerate}

\item Consider plane wave propagation in a homogeneous and transversely isotropic whole space. For such a material the elastic tensor takes the form
\begin{align*}
A_{ijkl} = {}&(A-2N)\delta_{ij}\delta_{kl} + N(\delta_{ik}\delta_{jl} + \delta_{il}\delta_{jk}) \\
&+ (F-A+2N)(\delta_{ij}\hat{\nu}_{k}\hat{\nu}_{l} + \hat{\nu}_{i}\hat{\nu}_{j}\delta_{kl})
+ (A+C-2F-4L)\hat{\nu}_{i}\hat{\nu}_{j}\hat{\nu}_{k}\hat{\nu}_{l} \\
&+ (L-N)(\hat{\nu}_{i}\hat{\nu}_{k}\delta_{jl} + \hat{\nu}_{j}\hat{\nu}_{k}\delta_{il} + \hat{\nu}_{j}\hat{\nu}_{l}\delta_{ik} + \hat{\nu}_{i}\hat{\nu}_{l}\delta_{jk}),
\end{align*}
where $A$, $C$, $F$, $L$ and $N$ are elastic moduli, and $\hat{\bm{\nu}}$ is a unit vector along the symmetry axis. Let $\theta$ denote the angle between the propagation direction, $\hat{\mathbf{p}}$, and the symmetry axis, and regard the departure of the medium from isotropy as small.
\begin{enumerate}
    \item Using first-order perturbation theory, obtain an expression for the phase speed of a quasi P-wave as a function of $\theta$.
    \item Using degenerate perturbation theory, obtain the first-order phase speeds of the two quasi S-waves as functions of $\theta$, and describe their polarisations. Check your results against the exact phase speeds for propagation along and perpendicular to the symmetry axis (these follow from the components $A_{3333}=C$, $A_{1313}=L$, $A_{1111}=A$ and $A_{1212}=N$ quoted in Lecture~14). [Hint: choose one of the unperturbed polarisation vectors to lie in the plane containing $\hat{\mathbf{p}}$ and $\hat{\bm{\nu}}$, and the other perpendicular to this plane.]
\end{enumerate}

\item Show that the time-dependent Schrödinger equation for a free particle
\begin{equation*}
    \ii\hbar\frac{\partial\psi}{\partial t}=-\frac{\hbar^{2}}{2m}\nabla^{2}\psi,
\end{equation*}
admits plane wave solutions of the form
\begin{equation*}
    \psi(\mathbf{x},t)=a\exp\left[\frac{-\ii(E t-\mathbf{p}\cdot\mathbf{x})}{\hbar}\right]
\end{equation*}
and obtain the relationship between the energy $E$ and momentum $\mathbf{p}$. Consider the corresponding equation for a particle in a non-constant but smoothly varying potential field $V$:
\begin{equation*}
    \ii\hbar\frac{\partial\psi}{\partial t}=\left(-\frac{\hbar^{2}}{2m}\nabla^{2}+V\right)\psi.
\end{equation*}
Starting from the ansatz
\begin{equation*}
    \psi(\mathbf{x},t)=a(\mathbf{x})\exp\left\{\frac{-\ii[E t-\varphi(\mathbf{x})]}{\hbar}\right\},
\end{equation*}
where the amplitude $a$ and phase function $\varphi$ are to be determined, and retaining only the leading-order terms in powers of $1/\hbar$, obtain the eikonal equation
\begin{equation*}
    H(\mathbf{x},\nabla\varphi)=E,
\end{equation*}
where $H$ is the classical Hamiltonian. Applying the method of characteristics, show that the particle's phase function can be determined through solution of the classical equations of motion.

\item Consider P-wave rays within an isotropic body. The travel time of a ray between two points $\mathbf{x}_{1}$ and $\mathbf{x}_{2}$ can be written
\begin{equation*}
    T = \int_{0}^{1} \frac{1}{\alpha(\mathbf{x})} \sqrt{\frac{\ddns x_{i}}{\ddns\gamma}\frac{\ddns x_{i}}{\ddns\gamma}} \dd \gamma,
\end{equation*}
where $\gamma\mapsto\mathbf{x}(\gamma)$ is the ray path between $\mathbf{x}_{1}$ and $\mathbf{x}_{2}$ parameterised by a generating parameter $\gamma\in[0,1]$. Regarding this travel time as a functional of the given ray path, Fermat's principle states that the true ray path is a stationary point of this functional. Obtain the corresponding Euler--Lagrange equations, and show that they are equivalent to the Hamiltonian ray tracing equations discussed within lectures. [Hint: it will be useful to express both sets of ordinary differential equations in terms of the arc length along the ray.]

\item Consider a plane wave $u_{i}=f(t-p_{j}x_{j})a_{i}$ within a homogeneous anisotropic body, where the slowness vector $\mathbf{p}$ and the unit polarisation vector $\mathbf{a}$ satisfy the Christoffel equation $\bm{\Gamma}(\mathbf{p})\mathbf{a}=\mathbf{a}$, with $\Gamma_{ik}(\mathbf{p})=\frac{1}{\rho}A_{ijkl}p_{j}p_{l}$. The energy conservation law of Question 2 carries over to the linearised theory if $T_{ij}$ and $v_{i}$ are replaced by their linearised forms, $A_{ijkl}\frac{\partial u_{k}}{\partial x_{l}}$ and $\frac{\partial u_{i}}{\partial t}$, and only terms of second order in the displacement are retained, giving the energy density and energy flux
\begin{equation*}
    E=\frac{1}{2}\rho\frac{\partial u_{i}}{\partial t}\frac{\partial u_{i}}{\partial t}+\frac{1}{2}A_{ijkl}\frac{\partial u_{i}}{\partial x_{j}}\frac{\partial u_{k}}{\partial x_{l}}, \quad s_{j}=-A_{ijkl}\frac{\partial u_{k}}{\partial x_{l}}\frac{\partial u_{i}}{\partial t}.
\end{equation*}
\begin{enumerate}
    \item Show that for the plane wave the kinetic and elastic contributions to $E$ are equal, and that $s_{j}=Ev_{j}$, where the \emph{energy velocity} is
    \begin{equation*}
        v_{j}=\frac{1}{\rho}A_{ijkl}a_{i}a_{k}p_{l}.
    \end{equation*}
    Show that $\mathbf{v}\cdot\hat{\mathbf{p}}=c$, the phase speed, and deduce that $\|\mathbf{v}\|\ge c$, with equality if and only if $\mathbf{v}$ is parallel to $\hat{\mathbf{p}}$. Evaluate $\mathbf{v}$ for P- and S-waves in an isotropic body.
    \hardp Let $\lambda(\mathbf{p})$ denote the eigenvalue of $\bm{\Gamma}(\mathbf{p})$ associated with $\mathbf{a}$, so that the corresponding sheet of the slowness surface is $\lambda(\mathbf{p})=1$ and the ray Hamiltonian for this sheet is $H(\mathbf{p})=\frac{1}{2}\lambda(\mathbf{p})$. Show that
    \begin{equation*}
        \frac{\partial\lambda}{\partial p_{j}}=\mathbf{a}\cdot\frac{\partial\bm{\Gamma}}{\partial p_{j}}\mathbf{a},
    \end{equation*}
    [Hint: differentiate $\lambda=\mathbf{a}\cdot\bm{\Gamma}\mathbf{a}$, remembering that $\mathbf{a}$ depends on $\mathbf{p}$ but has unit length.] Hence show that $\mathbf{v}=\partial H/\partial\mathbf{p}$, and deduce that the energy velocity is normal to the slowness surface and equal to $\ddns\mathbf{x}/\ddns\sigma$ along a ray.
    \hardp An impulsive point source acts at the origin of the body at time $t=0$. Using ray theory, show that the wavefront at time $t$ is the surface traced out by $t\,\mathbf{v}(\hat{\mathbf{p}})$ as $\hat{\mathbf{p}}$ ranges over the unit sphere; this is known as the \emph{wave surface}. Describe the geometric relationship between the wave surface and the slowness surface.
\end{enumerate}
\end{enumerate}
